cmake_minimum_required(VERSION 3.24)
project(EzConn VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network SerialPort Widgets)
find_package(ZLIB REQUIRED)
add_library(EasyConnCore src/core/Connection.h src/core/Connection.cpp src/core/LegacyProtocol.h src/core/LegacyProtocol.cpp src/core/ReadServices.h src/core/ReadServices.cpp)
target_include_directories(EasyConnCore PUBLIC src)
target_link_libraries(EasyConnCore PUBLIC Qt6::Core Qt6::Network Qt6::SerialPort)
target_sources(EasyConnCore PRIVATE src/core/ScreenService.h src/core/ScreenService.cpp)
target_sources(EasyConnCore PRIVATE src/core/ScannerService.h src/core/ScannerService.cpp)
target_link_libraries(EasyConnCore PUBLIC Qt6::Gui)
target_link_libraries(EasyConnCore PRIVATE ZLIB::ZLIB)
if(MSVC)
  target_compile_options(EasyConnCore PUBLIC /utf-8 /W4)
else()
  target_compile_options(EasyConnCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(EasyConn WIN32 src/gui/main.cpp)
target_link_libraries(EasyConn PRIVATE EasyConnCore Qt6::Widgets)
add_executable(EasyConnProbe src/probe/main.cpp)
target_link_libraries(EasyConnProbe PRIVATE EasyConnCore)
if(WIN32)
  add_custom_target(EasyConnLegacyHost ALL
    COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_SOURCE_DIR}/scripts/Build-LegacyHost.ps1" -OutputDir "$<TARGET_FILE_DIR:EasyConnProbe>/legacy"
    VERBATIM)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(EasyConnCoreTests tests/ConnectionTest.cpp)
  target_link_libraries(EasyConnCoreTests PRIVATE EasyConnCore Qt6::Test)
  add_test(NAME connection COMMAND EasyConnCoreTests)
  add_executable(EasyConnProtocolTests tests/ProtocolTest.cpp)
  target_link_libraries(EasyConnProtocolTests PRIVATE EasyConnCore Qt6::Test)
  if(WIN32)
    add_dependencies(EasyConnProtocolTests EasyConnLegacyHost)
  endif()
  add_test(NAME protocol COMMAND EasyConnProtocolTests)
  add_executable(EasyConnViewTests tests/ScreenViewTest.cpp)
  target_include_directories(EasyConnViewTests PRIVATE src)
  target_link_libraries(EasyConnViewTests PRIVATE EasyConnCore Qt6::Widgets Qt6::Test)
  add_test(NAME screen_view COMMAND EasyConnViewTests -platform offscreen)
endif()
